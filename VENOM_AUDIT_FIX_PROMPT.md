**Priority:** 🔴 Critical

```javascript
const crypto = require('crypto');

/**
 * Deduplicate findings by content fingerprint.
 * Groups identical findings and adds a `count` field.
 * Preserves the highest-severity instance.
 */
function deduplicateFindings(findings) {
  const map = new Map();

  for (const finding of findings) {
    const key = crypto
      .createHash('sha256')
      .update(`${(finding.title || '').trim()}::${(finding.description || '').trim()}`)
      .digest('hex')
      .substring(0, 16);

    if (!map.has(key)) {
      map.set(key, { ...finding, count: 1, _dedupKey: key });
    } else {
      const existing = map.get(key);
      existing.count += 1;
      // Keep highest severity instance
      const severityRank = { CRITICAL: 5, HIGH: 4, MEDIUM: 3, LOW: 2, INFO: 1 };
      const incomingRank = severityRank[(finding.severity || '').toUpperCase()] || 0;
      const existingRank = severityRank[(existing.severity || '').toUpperCase()] || 0;
      if (incomingRank > existingRank) {
        map.set(key, { ...finding, count: existing.count, _dedupKey: key });
      }
    }
  }

  return Array.from(map.values());
}

module.exports = { deduplicateFindings };
```

```javascript
function fingerprintFinding(finding) {

    .update(`${finding.title || ''}::${finding.description || ''}`)
    .digest('hex')
    .substring(0, 16);
}

async function createSnapshot(engagementId, snapshotType = 'scheduled') {
  const jobs = await ExecutionJob.find({ engagementId, status: 'completed' })
    .sort({ finishedAt: -1 })
    .limit(10)
    .lean();

  const allFindings = jobs.flatMap(j => j.output?.findings || []);
  const latestJob = jobs[0];

  return SecurityBaseline.create({
    engagementId,
    targetUrl: latestJob?.targetUrl,
    snapshotType,
    findings: allFindings.map(f => ({
      ...f,
      fingerprint: fingerprintFinding(f),
      firstSeenAt: new Date()
    })),
    openPorts: latestJob?.output?.ports || [],
    technologyStack: (latestJob?.output?.technologyFingerprint || '').split(', ').filter(Boolean)
  });
}

async function detectChanges(engagementId) {
  const snapshots = await SecurityBaseline.find({ engagementId })
    .sort({ snapshotAt: -1 })
    .limit(2)
    .lean();

  if (snapshots.length < 2) {
    return { changesFound: false, message: 'Need at least 2 snapshots to detect changes.' };
  }

  const [current, previous] = snapshots;
  const currentFps = new Set(current.findings.map(f => f.fingerprint));
  const previousFps = new Set(previous.findings.map(f => f.fingerprint));

  const newFindings = current.findings.filter(f => !previousFps.has(f.fingerprint));
  const resolvedFindings = previous.findings.filter(f => !currentFps.has(f.fingerprint));

  return {
    changesFound: newFindings.length > 0 || resolvedFindings.length > 0,
    newFindings,
    resolvedFindings,
    scanGapHours: Math.round((new Date(current.snapshotAt) - new Date(previous.snapshotAt)) / 3600000),
    currentSnapshotId: current._id,
    previousSnapshotId: previous._id
  };
}

module.exports = { createSnapshot, detectChanges };
```

### FIX 11 — CVSS SCORE DEFLATION FOR LOW-SIGNAL FINDINGS
**Priority:** 🟡 Medium  
**Files:** `backend/services/complianceMapper.js` → `computeOverallCvssScore()`  
**Why:** CVSS 5.5 for a single missing header finding is misleading. The 1.05 environmental multiplier inflates low findings. For findings with CVSS < 6.0, no multiplier should apply.

**Action:**

In `backend/services/complianceMapper.js`, replace `computeOverallCvssScore()`:

```javascript
function computeOverallCvssScore(findings) {
  const deduped = deduplicateFindings(findings);
  if (deduped.length === 0) return 0;

  const scores = deduped.map(f => parseFloat(f.cvssScore) || 0).filter(s => s > 0);
  if (scores.length === 0) return 0;

  const maxScore = Math.max(...scores);

  // Only apply environmental modifier for genuinely high/critical findings
  // Low/medium findings: report actual CVSS, no inflation
  const modifier = maxScore >= 7.0 ? 1.05 : 1.0;

  return Math.min(10, parseFloat((maxScore * modifier).toFixed(1)));
}
```
